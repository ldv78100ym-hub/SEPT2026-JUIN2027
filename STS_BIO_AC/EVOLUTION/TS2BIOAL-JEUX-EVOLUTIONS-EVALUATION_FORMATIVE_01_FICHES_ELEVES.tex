\begin{fiche}{Rappels utiles pour le correcteur}
\textbullet{} \textbf{Ne pas accepter} la somme des taux pour un taux global : exiger le passage par les coefficients multiplicateurs.\\[0.2cm]
\textbullet{} \textbf{Ne pas pénaliser} une erreur de calcul final si la démarche est correcte (sauf si le résultat est manifestement incohérent).\\[0.2cm]
\textbullet{} \textbf{Valoriser} toute phrase de conclusion, même courte, qui interprète le résultat ($+$ ou $-$).\\[0.2cm]
\textbullet{} \textbf{Signaler} aux équipes les plus rapides qu'elles peuvent viser la 4\textsuperscript{e} enveloppe (4 pts) plutôt que de peaufiner la 1\textsuperscript{re} (2 pts).\\[0.2cm]
\textbullet{} \textbf{En cas d'égalité}, poser la question bonus : \textit{un prix augmente de $100\%$ puis baisse de $50\%$ ; revient-il au prix initial ?} (Réponse : \textbf{oui}, car $2 \times 0{,}5 = 1$.)
\end{fiche}
